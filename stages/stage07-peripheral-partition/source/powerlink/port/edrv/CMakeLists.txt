cmake_minimum_required(VERSION 3.16)
project(m7_powerlink_eth2_edrv C)
if(NOT EXISTS "${OPLK_SOURCE}/stack/include/kernel/edrv.h")
    message(FATAL_ERROR "Set OPLK_SOURCE to the prepared, pinned upstream tree")
endif()
set(port "${CMAKE_CURRENT_SOURCE_DIR}/..")
add_library(m7_powerlink_eth2_edrv STATIC "${port}/edrv-rk3572.c")
target_include_directories(m7_powerlink_eth2_edrv PUBLIC
    "${port}" "${OPLK_SOURCE}/stack/include" "${OPLK_SOURCE}/stack/src" "${OPLK_SOURCE}/contrib")
target_compile_definitions(m7_powerlink_eth2_edrv PUBLIC OPLK_TARGET_UNIPROTON CONFIG_MN NDEBUG)
target_compile_options(m7_powerlink_eth2_edrv PRIVATE
    -Wall -Wextra -Werror -fno-strict-aliasing -ffunction-sections -fdata-sections)
set_target_properties(m7_powerlink_eth2_edrv PROPERTIES C_STANDARD 99 C_STANDARD_REQUIRED YES)
if(CMAKE_SYSTEM_PROCESSOR STREQUAL "aarch64")
    target_compile_options(m7_powerlink_eth2_edrv PUBLIC
        -mcpu=cortex-a53 -mgeneral-regs-only -mstrict-align -mno-outline-atomics)
endif()
if(NOT CMAKE_CROSSCOMPILING)
    enable_testing()
    target_compile_definitions(m7_powerlink_eth2_edrv PRIVATE M7_EDRV_NATIVE_TEST)
    add_executable(test_powerlink_edrv "${port}/../../../tests/test_powerlink_edrv.c")
    target_link_libraries(test_powerlink_edrv PRIVATE m7_powerlink_eth2_edrv)
    target_compile_options(test_powerlink_edrv PRIVATE -UNDEBUG -Wall -Wextra -Werror)
    foreach(test init buffers tx_wrap tx_callback tx_error stop rx_filters rx_invalid rx_deferred)
        add_test(NAME edrv_${test} COMMAND test_powerlink_edrv ${test})
    endforeach()
endif()
