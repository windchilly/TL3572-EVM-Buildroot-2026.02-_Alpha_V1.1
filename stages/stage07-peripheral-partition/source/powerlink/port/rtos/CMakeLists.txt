cmake_minimum_required(VERSION 3.16)
project(m7_powerlink_rtos C)
if(NOT EXISTS "${OPLK_SOURCE}/stack/include/kernel/hrestimer.h")
    message(FATAL_ERROR "Set OPLK_SOURCE to the prepared pinned upstream")
endif()
set(port "${CMAKE_CURRENT_SOURCE_DIR}/..")
add_library(m7_powerlink_rtos STATIC "${port}/target-uniproton.c" "${port}/hrestimer-rk3572.c")
target_include_directories(m7_powerlink_rtos PUBLIC "${port}" "${OPLK_SOURCE}/stack/include" "${OPLK_SOURCE}/contrib")
target_compile_definitions(m7_powerlink_rtos PUBLIC OPLK_TARGET_UNIPROTON CONFIG_MN NDEBUG)
target_compile_options(m7_powerlink_rtos PRIVATE -Wall -Wextra -Werror -fno-strict-aliasing -ffunction-sections -fdata-sections)
set_target_properties(m7_powerlink_rtos PROPERTIES C_STANDARD 99 C_STANDARD_REQUIRED YES)
if(CMAKE_SYSTEM_PROCESSOR STREQUAL "aarch64")
    target_compile_options(m7_powerlink_rtos PUBLIC -mcpu=cortex-a53 -mgeneral-regs-only -mstrict-align -mno-outline-atomics)
endif()
if(NOT CMAKE_CROSSCOMPILING)
    enable_testing()
    add_executable(test_powerlink_rtos "${port}/../../../tests/test_powerlink_rtos.c")
    target_link_libraries(test_powerlink_rtos PRIVATE m7_powerlink_rtos)
    target_compile_options(test_powerlink_rtos PRIVATE -UNDEBUG -Wall -Wextra -Werror)
    foreach(test conversion lifecycle expiry periodic callback slow_callback stale fault mutex critical cache owner)
        add_test(NAME rtos_${test} COMMAND test_powerlink_rtos ${test})
    endforeach()
endif()
